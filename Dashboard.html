Arduino Code for Sensor Reading

// Jorethang Smart Farming Arduino Code
// Soil moisture, temperature and humidity monitoring

// Define sensor pins
#define MOISTURE_SENSOR A0
#define TEMP_SENSOR A1
#define DHTPIN 2
#define DHTTYPE DHT11

// Include libraries
#include 

// Initialize DHT sensor
DHT dht(DHTPIN, DHTTYPE);

void setup() {
  // Start serial communication
  Serial.begin(9600);
  
  // Initialize DHT sensor
  dht.begin();
  
  // Set pin modes
  pinMode(MOISTURE_SENSOR, INPUT);
  pinMode(TEMP_SENSOR, INPUT);
  
  Serial.println("Jorethang Smart Farming System Started");
}

void loop() {
  // Read soil moisture
  int moistureValue = analogRead(MOISTURE_SENSOR);
  int moisturePercent = map(moistureValue, 0, 1023, 0, 100);
  
  // Read temperature
  float temperature = dht.readTemperature();
  
  // Read humidity
  float humidity = dht.readHumidity();
  
  // Print values to serial
  Serial.print("Moisture:");
  Serial.print(moisturePercent);
  Serial.print("%,Temperature:");
  Serial.print(temperature);
  Serial.print("C,Humidity:");
  Serial.print(humidity);
  Serial.println("%");
  
  // Wait before next reading
  delay(2000);
}
